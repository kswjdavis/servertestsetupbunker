```python
from cachetools import TTLCache

weather_cache = TTLCache(maxsize=1, ttl=300)  # 5 min cache

async def fetch_weather():
    if "weather" in weather_cache:
        return weather_cache["weather"]

    data = await http_client.get(WEATHER_API_URL)
    weather_cache["weather"] = data
    return data
```

**Query Optimization (AC3):**
```python
# server/app/repositories/device_repository.py
async def get_bunker_fan_status(self, bunker_id: UUID):
    # Use select_related to avoid N+1 queries
    result = await self.db.execute(
        select(Device)
        .options(selectinload(Device.status))  # Eager load
        .where(Device.bunker_id == bunker_id)
    )
    return result.scalars().all()
```

### Load Testing (AC9)

**Simulate 10 Devices:**
```python
# scripts/load_test.py
import asyncio
import aiohttp

async def simulate_device(device_id: int):
    async with aiohttp.ClientSession() as session:
        while True:
            await session.post(
                "https://server.com/api/v1/control/status",
                json={"device_id": device_id, "relay_state": "ON"},
                headers={"Authorization": f"Bearer {TOKEN}"}
            )
            await asyncio.sleep(60)  # Report every 60s

async def main():
    tasks = [simulate_device(i) for i in range(10)]
    await asyncio.gather(*tasks)

asyncio.run(main())
```
