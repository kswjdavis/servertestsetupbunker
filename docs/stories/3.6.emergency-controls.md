# Story 3.6: Emergency ON Controls

## Story
**As a** web UI operator,
**I want** prominent emergency ON buttons to immediately activate all fans,
**so that** I can respond quickly to severe weather or safety concerns (FR38, FR39).

### API Endpoints

**POST /api/v1/control/emergency-on**
[Source: architecture.md#REST API Specification - Device Control]

**Request Body:**
```typescript
{
  bunker_id: string;  // UUID of specific bunker
}
```

**Response:**
```typescript
{
  success: true,
  message: "Emergency ON activated for bunker",
  bunker_id: string,
  affected_devices: number
}
```

**POST /api/v1/control/emergency-on-all**

**Response:**
```typescript
{
  success: true,
  message: "Emergency ON activated for all bunkers",
  affected_bunkers: number,
  affected_devices: number
}
```

**DELETE /api/v1/control/emergency-on** or **emergency-on-all**

Clears emergency mode (AC: 9)

### Component Pattern

Follow standard React + TypeScript component pattern:
[Source: architecture.md#Frontend Architecture - Component Template Example]

```typescript
// components/emergency/EmergencyButton.tsx
import React, { useState } from 'react';

interface EmergencyButtonProps {
  scope: 'global' | 'bunker';
  bunkerId?: string;
  onSuccess?: () => void;
}

export const EmergencyButton: React.FC<EmergencyButtonProps> = ({ scope, bunkerId, onSuccess }) => {
  const [showModal, setShowModal] = useState(false);
  const [loading, setLoading] = useState(false);

  const handleConfirm = async () => {
    setLoading(true);
    try {
      if (scope === 'global') {
        await controlService.emergencyOnAll();
      } else {
        await controlService.emergencyOnBunker(bunkerId!);
      }
      setShowModal(false);
      toast.success('All fans activated');
      onSuccess?.();
    } catch (error) {
      toast.error('Failed to activate emergency mode');
    } finally {
      setLoading(false);
    }
  };

  return (
    <>
      <button
        onClick={() => setShowModal(true)}
        className="bg-red-600 hover:bg-red-700 text-white font-bold py-3 px-6 rounded-lg shadow-lg min-h-[44px] min-w-[44px]"
      >
        {scope === 'global' ? 'EMERGENCY ON - ALL BUNKERS' : 'EMERGENCY ON'}
      </button>

      {showModal && (
        <EmergencyModal
          scope={scope}
          onConfirm={handleConfirm}
          onCancel={() => setShowModal(false)}
          loading={loading}
        />
      )}
    </>
  );
};
```

### Service Layer Implementation

```typescript
// services/control.service.ts
import api from './api';

export async function emergencyOnAll(): Promise<EmergencyOnResponse> {
  const response = await api.post<EmergencyOnResponse>('/control/emergency-on-all');
  return response.data;
}

export async function emergencyOnBunker(bunkerId: string): Promise<EmergencyOnResponse> {
  const response = await api.post<EmergencyOnResponse>('/control/emergency-on', { bunker_id: bunkerId });
  return response.data;
}

export async function clearEmergencyAll(): Promise<void> {
  await api.delete('/control/emergency-on-all');
}

export async function clearEmergencyBunker(bunkerId: string): Promise<void> {
  await api.delete(`/control/emergency-on/${bunkerId}`);
}
```
