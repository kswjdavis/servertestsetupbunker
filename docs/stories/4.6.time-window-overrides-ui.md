# Story 4.6: Time Window Overrides UI

## Status
Done

## Story
**As a** system administrator,
**I want** to schedule time windows when fans must stay ON,
**so that** I can force fans ON during critical periods like harvest (FR36).

## Acceptance Criteria

1. Time window overrides list on bunker detail page
2. "Add Override" button opens form
3. Form fields: Start date/time, End date/time, Reason, Scope (this bunker or all bunkers)
4. Validation: end time must be after start time
5. Active overrides highlighted (currently in effect)
6. Past overrides grayed out or hidden
7. Delete override button
8. Calls POST /api/v1/overrides (create), DELETE /api/v1/overrides/:id
9. Poll GET /api/v1/overrides every 30 seconds
10. Timezone handling (use browser timezone or UTC with conversion)

### API Endpoints

[Source: architecture.md#REST API Specification]

**GET /api/v1/overrides**
Query params: `?bunker_id=<uuid>` (optional)
Response:
```typescript
[
  {
    id: string;              // UUID
    bunker_id: string | null; // null = global override
    start_time: string;      // ISO 8601 (UTC)
    end_time: string;        // ISO 8601 (UTC)
    reason: string;
    created_by: string;      // UUID of user
    created_at: string;      // ISO 8601
    is_active: boolean;      // Computed: now between start and end
  }
]
```

**POST /api/v1/overrides**
Request:
```typescript
{
  bunker_id: string | null;  // null = all bunkers
  start_time: string;        // ISO 8601 (UTC)
  end_time: string;          // ISO 8601 (UTC)
  reason: string;            // Max 500 chars
}
```
Response: 201 Created with created override

**DELETE /api/v1/overrides/:id**
Response: 204 No Content

```typescript
// components/bunker/TimeWindowOverridesList.tsx
import React, { useState } from 'react';
import { TimeWindowOverride } from '@/types';
import { useTimeWindowOverrides } from '@/hooks/useTimeWindowOverrides';
import { TimeWindowOverrideForm } from './TimeWindowOverrideForm';
import { TimeWindowOverrideCard } from './TimeWindowOverrideCard';
import * as overrideService from '@/services/override.service';

interface TimeWindowOverridesListProps {
  bunkerId: string;
  bunkerName: string;
}

export const TimeWindowOverridesList: React.FC<TimeWindowOverridesListProps> = ({
  bunkerId,
  bunkerName
}) => {
  const { data: overrides, loading, refresh } = useTimeWindowOverrides(bunkerId, 30000);
  const [showForm, setShowForm] = useState(false);
  const [isDeleting, setIsDeleting] = useState<string | null>(null);

  const handleDelete = async (overrideId: string) => {
    if (!confirm('Delete this time window override?')) return;

    setIsDeleting(overrideId);
    try {
      await overrideService.deleteOverride(overrideId);
      refresh();
      // Show success toast
    } catch (error) {
      console.error('Failed to delete override:', error);
      // Show error toast
    } finally {
      setIsDeleting(null);
    }
  };

  const handleCreateSuccess = () => {
    setShowForm(false);
    refresh();
    // Show success toast
  };

  if (loading) return <div>Loading overrides...</div>;

  // Separate active and past overrides
  const activeOverrides = overrides?.filter(o => o.is_active) || [];
  const pastOverrides = overrides?.filter(o => !o.is_active) || [];

  return (
    <div className="bg-white rounded-lg shadow-lg p-6">
      <div className="flex justify-between items-center mb-4">
        <h3 className="text-xl font-semibold">Time Window Overrides</h3>
        <button
          onClick={() => setShowForm(true)}
          className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700"
        >
          Add Override
        </button>
      </div>

      <p className="text-sm text-gray-600 mb-6">
        Schedule periods when fans must stay ON regardless of wind conditions.
      </p>

      {/* Active Overrides */}
      {activeOverrides.length > 0 && (
        <div className="mb-6">
          <h4 className="text-lg font-medium text-green-700 mb-3">Active Overrides</h4>
          <div className="space-y-3">
            {activeOverrides.map(override => (
              <TimeWindowOverrideCard
                key={override.id}
                override={override}
                isActive={true}
                onDelete={() => handleDelete(override.id)}
                isDeleting={isDeleting === override.id}
              />
            ))}
          </div>
        </div>
      )}

      {/* Past Overrides */}
      {pastOverrides.length > 0 && (
        <div>
          <h4 className="text-lg font-medium text-gray-600 mb-3">Past Overrides</h4>
          <div className="space-y-3 opacity-60">
            {pastOverrides.slice(0, 5).map(override => (
              <TimeWindowOverrideCard
                key={override.id}
                override={override}
                isActive={false}
                onDelete={() => handleDelete(override.id)}
                isDeleting={isDeleting === override.id}
              />
            ))}
          </div>
        </div>
      )}

      {overrides && overrides.length === 0 && (
        <div className="text-center py-8 text-gray-500">
          No time window overrides configured
        </div>
      )}

      {/* Create Override Modal */}
      {showForm && (
        <TimeWindowOverrideForm
          bunkerId={bunkerId}
          bunkerName={bunkerName}
          onClose={() => setShowForm(false)}
          onSuccess={handleCreateSuccess}
        />
      )}
    </div>
  );
};
```

### Override Form Modal

```typescript
// components/bunker/TimeWindowOverrideForm.tsx
import React, { useState } from 'react';
import { TimeWindowOverrideFormData, TimeWindowOverrideCreateRequest } from '@/types';
import * as overrideService from '@/services/override.service';
import { convertLocalToUTC } from '@/utils/timezone';

interface TimeWindowOverrideFormProps {
  bunkerId: string;
  bunkerName: string;
  onClose: () => void;
  onSuccess: () => void;
}

export const TimeWindowOverrideForm: React.FC<TimeWindowOverrideFormProps> = ({
  bunkerId,
  bunkerName,
  onClose,
  onSuccess
}) => {
  const [formData, setFormData] = useState<TimeWindowOverrideFormData>({
    scope: 'bunker',
    startDate: '',
    startTime: '',
    endDate: '',
    endTime: '',
    reason: ''
  });
  const [errors, setErrors] = useState<Record<string, string>>({});
  const [isSubmitting, setIsSubmitting] = useState(false);

  const validate = (): boolean => {
    const newErrors: Record<string, string> = {};

    if (!formData.startDate) newErrors.startDate = 'Start date is required';
    if (!formData.startTime) newErrors.startTime = 'Start time is required';
    if (!formData.endDate) newErrors.endDate = 'End date is required';
    if (!formData.endTime) newErrors.endTime = 'End time is required';
    if (!formData.reason.trim()) newErrors.reason = 'Reason is required';

    // Validate end time is after start time
    if (formData.startDate && formData.startTime && formData.endDate && formData.endTime) {
      const start = new Date(`${formData.startDate}T${formData.startTime}`);
      const end = new Date(`${formData.endDate}T${formData.endTime}`);
      if (end <= start) {
        newErrors.endDate = 'End time must be after start time';
      }
    }

    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();

    if (!validate()) return;

    setIsSubmitting(true);

    try {
      // Convert local datetime to UTC ISO 8601
      const startUTC = convertLocalToUTC(`${formData.startDate}T${formData.startTime}`);
      const endUTC = convertLocalToUTC(`${formData.endDate}T${formData.endTime}`);

      const payload: TimeWindowOverrideCreateRequest = {
        bunker_id: formData.scope === 'bunker' ? bunkerId : null,
        start_time: startUTC,
        end_time: endUTC,
        reason: formData.reason.trim()
      };

      await overrideService.createOverride(payload);
      onSuccess();
    } catch (error) {
      console.error('Failed to create override:', error);
      // Show error toast
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50">
      <div className="bg-white rounded-lg p-6 max-w-lg w-full mx-4 max-h-screen overflow-y-auto">
        <h3 className="text-xl font-bold mb-4">Add Time Window Override</h3>

        <form onSubmit={handleSubmit} className="space-y-4">
          {/* Scope Selector */}
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">
              Apply To
            </label>
            <div className="space-y-2">
              <label className="flex items-center space-x-2">
                <input
                  type="radio"
                  name="scope"
                  value="bunker"
                  checked={formData.scope === 'bunker'}
                  onChange={(e) => setFormData({ ...formData, scope: 'bunker' })}
                  className="w-4 h-4"
                />
                <span>This bunker only ({bunkerName})</span>
              </label>
              <label className="flex items-center space-x-2">
                <input
                  type="radio"
                  name="scope"
                  value="global"
                  checked={formData.scope === 'global'}
                  onChange={(e) => setFormData({ ...formData, scope: 'global' })}
                  className="w-4 h-4"
                />
                <span>All bunkers (global override)</span>
              </label>
            </div>
          </div>

          {/* Start Date/Time */}
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">
              Start Date & Time *
            </label>
            <div className="grid grid-cols-2 gap-2">
              <input
                type="date"
                value={formData.startDate}
                onChange={(e) => setFormData({ ...formData, startDate: e.target.value })}
                className="px-3 py-2 border rounded-lg"
              />
              <input
                type="time"
                value={formData.startTime}
                onChange={(e) => setFormData({ ...formData, startTime: e.target.value })}
                className="px-3 py-2 border rounded-lg"
              />
            </div>
            {errors.startDate && <p className="text-red-500 text-sm mt-1">{errors.startDate}</p>}
            {errors.startTime && <p className="text-red-500 text-sm mt-1">{errors.startTime}</p>}
          </div>

          {/* End Date/Time */}
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">
              End Date & Time *
            </label>
            <div className="grid grid-cols-2 gap-2">
              <input
                type="date"
                value={formData.endDate}
                onChange={(e) => setFormData({ ...formData, endDate: e.target.value })}
                className="px-3 py-2 border rounded-lg"
              />
              <input
                type="time"
                value={formData.endTime}
                onChange={(e) => setFormData({ ...formData, endTime: e.target.value })}
                className="px-3 py-2 border rounded-lg"
              />
            </div>
            {errors.endDate && <p className="text-red-500 text-sm mt-1">{errors.endDate}</p>}
            {errors.endTime && <p className="text-red-500 text-sm mt-1">{errors.endTime}</p>}
          </div>

          {/* Reason */}
          <div>
            <label className="block text-sm font-medium text-gray-700 mb-2">
              Reason *
            </label>
            <textarea
              value={formData.reason}
              onChange={(e) => setFormData({ ...formData, reason: e.target.value })}
              placeholder="e.g., Harvest operations, Maintenance window"
              maxLength={500}
              rows={3}
              className="w-full px-3 py-2 border rounded-lg focus:ring-2 focus:ring-blue-500"
            />
            {errors.reason && <p className="text-red-500 text-sm mt-1">{errors.reason}</p>}
            <p className="text-xs text-gray-500 mt-1">
              {formData.reason.length}/500 characters
            </p>
          </div>

          {/* Action Buttons */}
          <div className="flex justify-end space-x-4 pt-4">
            <button
              type="button"
              onClick={onClose}
              className="px-4 py-2 border border-gray-300 rounded-lg hover:bg-gray-50"
            >
              Cancel
            </button>
            <button
              type="submit"
              disabled={isSubmitting}
              className="px-4 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 disabled:opacity-50"
            >
              {isSubmitting ? 'Creating...' : 'Create Override'}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
};
```

```typescript
// services/override.service.ts
import api from './api';
import { TimeWindowOverride, TimeWindowOverrideCreateRequest } from '@/types';

export async function getOverrides(bunkerId?: string): Promise<TimeWindowOverride[]> {
  const url = bunkerId ? `/overrides?bunker_id=${bunkerId}` : '/overrides';
  const response = await api.get<TimeWindowOverride[]>(url);
  return response.data;
}

export async function createOverride(
  data: TimeWindowOverrideCreateRequest
): Promise<TimeWindowOverride> {
  const response = await api.post<TimeWindowOverride>('/overrides', data);
  return response.data;
}

export async function deleteOverride(id: string): Promise<void> {
  await api.delete(`/overrides/${id}`);
}
```

### Timezone Utilities

```typescript
// utils/timezone.ts

/**
 * Convert local datetime string to UTC ISO 8601
 * @param localDateTime - Local datetime string (YYYY-MM-DDTHH:MM)
 * @returns UTC ISO 8601 string
 */
export function convertLocalToUTC(localDateTime: string): string {
  const localDate = new Date(localDateTime);
  return localDate.toISOString();
}

/**
 * Format UTC ISO 8601 string to local datetime display
 * @param utcString - UTC ISO 8601 string
 * @returns Formatted local datetime string
 */
export function formatDateTime(utcString: string): string {
  const date = new Date(utcString);
  return date.toLocaleString('en-US', {
    year: 'numeric',
    month: 'short',
    day: 'numeric',
    hour: '2-digit',
    minute: '2-digit',
    hour12: true
  });
}

/**
 * Convert UTC ISO 8601 to local date input format (YYYY-MM-DD)
 */
export function toLocalDateInput(utcString: string): string {
  const date = new Date(utcString);
  return date.toISOString().split('T')[0];
}

/**
 * Convert UTC ISO 8601 to local time input format (HH:MM)
 */
export function toLocalTimeInput(utcString: string): string {
  const date = new Date(utcString);
  return date.toTimeString().slice(0, 5);
}
```

### Custom Polling Hook

```typescript
// hooks/useTimeWindowOverrides.ts
import { useState, useEffect } from 'react';
import { TimeWindowOverride } from '@/types';
import * as overrideService from '@/services/override.service';

export function useTimeWindowOverrides(bunkerId: string, pollInterval: number = 30000) {
  const [data, setData] = useState<TimeWindowOverride[] | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  const fetchOverrides = async () => {
    try {
      const overrides = await overrideService.getOverrides(bunkerId);
      setData(overrides);
      setError(null);
    } catch (err) {
      setError(err as Error);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchOverrides(); // Initial fetch
    const interval = setInterval(fetchOverrides, pollInterval);
    return () => clearInterval(interval);
  }, [bunkerId, pollInterval]);

  return { data, loading, error, refresh: fetchOverrides };
}
```
