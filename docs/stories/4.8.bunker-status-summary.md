### API Integration
Reference: **docs/architecture.md#Backend Architecture**

**Endpoints Used:**
- `GET /api/v1/bunkers` - List all bunkers with summary data
- `GET /api/v1/bunkers/{bunker_id}/status` - Detailed status per bunker (for real-time updates)

**Response Schema (BunkerSummary):**
```typescript
interface BunkerSummary {
  id: string;                    // UUID
  name: string;                  // Bunker name
  location: string;              // Computed from lat/lng or stored separately
  fans_online: number;           // Count of online fans
  fans_total: number;            // Total fans in bunker
  fans_running: number;          // Fans currently ON
  energy_saved_today: number;    // kWh saved today
  status: 'ok' | 'warning' | 'critical';  // Computed status
  last_update: string;           // ISO 8601 timestamp
}
```

**Status Logic (for AC3):**
```typescript
function computeStatus(summary: BunkerSummary): 'ok' | 'warning' | 'critical' {
  const offlineCount = summary.fans_total - summary.fans_online;

  if (offlineCount === 0) {
    return 'ok';  // All fans online - green
  } else if (offlineCount <= summary.fans_total * 0.3) {
    return 'warning';  // <30% offline - yellow
  } else {
    return 'critical';  // >30% offline - red
  }
}
```

```typescript
import React from 'react';
import { useNavigate } from 'react-router-dom';

interface BunkerCardProps {
  bunker: BunkerSummary;

      {/* Fan Status (AC2) */}
```

**DashboardPage.tsx** (AC5, AC6, AC7, AC8, AC9):
```typescript
// pages/DashboardPage.tsx
import React, { useState, useMemo } from 'react';
import { BunkerCard } from '@/components/bunker/BunkerCard';
import { BunkerCardSkeleton } from '@/components/bunker/BunkerCardSkeleton';
import { useBunkers } from '@/hooks/useBunkers';

type SortOption = 'name' | 'savings' | 'status';

export const DashboardPage: React.FC = () => {
  const [sortBy, setSortBy] = useState<SortOption>('name');

  // AC9: Poll every 5 seconds
  const { bunkers, loading, error } = useBunkers(5000);

  // AC8: Sorting logic
  const sortedBunkers = useMemo(() => {
    const bunkersCopy = [...bunkers];

    switch (sortBy) {
      case 'name':
        return bunkersCopy.sort((a, b) => a.name.localeCompare(b.name));
      case 'savings':
        return bunkersCopy.sort((a, b) => b.energy_saved_today - a.energy_saved_today);
      case 'status':
        const statusOrder = { critical: 0, warning: 1, ok: 2 };
        return bunkersCopy.sort((a, b) => statusOrder[a.status] - statusOrder[b.status]);
      default:
        return bunkersCopy;
    }
  }, [bunkers, sortBy]);

  // AC7: Empty state
  if (!loading && bunkers.length === 0) {
    return (
      <div className="flex flex-col items-center justify-center h-96">
        <p className="text-gray-500 text-lg mb-4">No bunkers configured</p>
        <button className="btn-primary">Add First Bunker</button>
      </div>
    );
  }

  return (
    <div className="container mx-auto px-4 py-8">
      {/* Header with Sort Controls (AC8) */}
      <div className="flex justify-between items-center mb-6">
        <h1 className="text-3xl font-bold">Bunker Dashboard</h1>
        <div className="flex gap-2">
          <button
            onClick={() => setSortBy('name')}
            className={`px-4 py-2 rounded ${sortBy === 'name' ? 'bg-blue-500 text-white' : 'bg-gray-200'}`}
          >
            Name
          </button>
          <button
            onClick={() => setSortBy('savings')}
            className={`px-4 py-2 rounded ${sortBy === 'savings' ? 'bg-blue-500 text-white' : 'bg-gray-200'}`}
          >
            Savings
          </button>
          <button
            onClick={() => setSortBy('status')}
            className={`px-4 py-2 rounded ${sortBy === 'status' ? 'bg-blue-500 text-white' : 'bg-gray-200'}`}
          >
            Status
          </button>
        </div>
      </div>

      {/* AC5: Responsive Grid Layout */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-6">
        {loading ? (
          // AC6: Loading skeletons
          Array.from({ length: 6 }).map((_, i) => <BunkerCardSkeleton key={i} />)
        ) : (
          sortedBunkers.map((bunker) => (
            <BunkerCard key={bunker.id} bunker={bunker} />
          ))
        )}
      </div>

      {error && (
        <div className="mt-4 p-4 bg-red-100 border border-red-400 text-red-700 rounded">
          Error loading bunkers: {error.message}
        </div>
      )}
    </div>
  );
};
```

**bunker.service.ts** (UPDATE):
```typescript
// services/bunker.service.ts
import api from './api';
import { Bunker, BunkerSummary } from '@/types';

export async function listBunkerSummaries(): Promise<BunkerSummary[]> {
  const response = await api.get<BunkerSummary[]>('/bunkers/summaries');
  return response.data;
}

// Alternatively, compute from full bunker + status data
export async function getBunkerSummary(bunkerId: string): Promise<BunkerSummary> {
  const [bunker, status] = await Promise.all([
    api.get<Bunker>(`/bunkers/${bunkerId}`),
    api.get(`/bunkers/${bunkerId}/status`)
  ]);

  return {
    id: bunker.data.id,
    name: bunker.data.name,
    location: `${bunker.data.latitude.toFixed(4)}, ${bunker.data.longitude.toFixed(4)}`,
    fans_online: status.data.aggregate.online_fans,
    fans_total: status.data.aggregate.total_fans,
    fans_running: status.data.aggregate.fans_running,
    energy_saved_today: 0,  // TODO: Calculate from energy endpoint
    status: computeStatus(status.data),
    last_update: new Date().toISOString()
  };
}
```

```python
@router.get("/summaries", response_model=List[BunkerSummary])
async def list_bunker_summaries(

    bunker_repo = BunkerRepository(db)

    bunkers = await bunker_repo.list()

    for bunker in bunkers:
```
