## Story

**I want** UI to create, edit, and delete bunkers,

### API Endpoints

[Source: architecture.md#REST API Specification]

**POST /api/v1/bunkers**
Request:
```typescript
{
  name: string;                // Required, max 100 chars
  latitude: number;            // Required, -90 to 90
  longitude: number;           // Required, -180 to 180
  orientation_degrees: number; // Required, 0-360
  fan_count: number;           // Required, >= 1
  electricity_cost_kwh: number; // Required, >= 0
  fan_power_watts: number;     // Required, >= 0
}
```
Response: 201 Created
```typescript
{
  id: string;  // UUID of created bunker
  // ... all bunker fields
}
```

**PUT /api/v1/bunkers/:id**
Request: Same as POST
Response: 200 OK with updated bunker

**DELETE /api/v1/bunkers/:id**
Response: 204 No Content

### Form Component Implementation

[Source: architecture.md#Frontend Architecture - Component Template Example]

```typescript
// components/bunker/BunkerForm.tsx
import React, { useState, useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import { Bunker, BunkerCreateRequest, BunkerFormData, BunkerFormErrors } from '@/types';
import * as bunkerService from '@/services/bunker.service';
import { MapLocationPicker } from './MapLocationPicker';
import { OrientationSelector } from './OrientationSelector';

interface BunkerFormProps {
  mode: 'create' | 'edit';
  initialData?: Bunker;
  onSuccess?: () => void;
}

export const BunkerForm: React.FC<BunkerFormProps> = ({ mode, initialData, onSuccess }) => {
  const navigate = useNavigate();
  const [formData, setFormData] = useState<BunkerFormData>({
    name: initialData?.name || '',
    latitude: initialData?.latitude.toString() || '',
    longitude: initialData?.longitude.toString() || '',
    orientation_degrees: initialData?.orientation_degrees.toString() || '0',
    fan_count: initialData?.fan_count.toString() || '1',
    electricity_cost_kwh: initialData?.electricity_cost_kwh.toString() || '0.12',
    fan_power_watts: initialData?.fan_power_watts.toString() || '500'
  });
  const [errors, setErrors] = useState<BunkerFormErrors>({});
  const [isSubmitting, setIsSubmitting] = useState(false);

  const validateForm = (): boolean => {
    const newErrors: BunkerFormErrors = {};

    if (!formData.name.trim()) {
      newErrors.name = 'Bunker name is required';
    } else if (formData.name.length > 100) {
      newErrors.name = 'Name must be 100 characters or less';
    }

    const lat = parseFloat(formData.latitude);
    if (isNaN(lat) || lat < -90 || lat > 90) {
      newErrors.latitude = 'Latitude must be between -90 and 90';
    }

    const lng = parseFloat(formData.longitude);
    if (isNaN(lng) || lng < -180 || lng > 180) {
      newErrors.longitude = 'Longitude must be between -180 and 180';
    }

    const orientation = parseFloat(formData.orientation_degrees);
    if (isNaN(orientation) || orientation < 0 || orientation >= 360) {
      newErrors.orientation_degrees = 'Orientation must be between 0 and 359';
    }

    const fanCount = parseInt(formData.fan_count);
    if (isNaN(fanCount) || fanCount < 1) {
      newErrors.fan_count = 'Fan count must be at least 1';
    }

    const cost = parseFloat(formData.electricity_cost_kwh);
    if (isNaN(cost) || cost < 0) {
      newErrors.electricity_cost_kwh = 'Cost must be a positive number';
    }

    const power = parseInt(formData.fan_power_watts);
    if (isNaN(power) || power < 0) {
      newErrors.fan_power_watts = 'Power must be a positive number';
    }

    setErrors(newErrors);
    return Object.keys(newErrors).length === 0;
  };

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();

    if (!validateForm()) return;

    setIsSubmitting(true);

    try {
      const payload: BunkerCreateRequest = {
        name: formData.name.trim(),
        latitude: parseFloat(formData.latitude),
        longitude: parseFloat(formData.longitude),
        orientation_degrees: parseFloat(formData.orientation_degrees),
        fan_count: parseInt(formData.fan_count),
        electricity_cost_kwh: parseFloat(formData.electricity_cost_kwh),
        fan_power_watts: parseInt(formData.fan_power_watts)
      };

      if (mode === 'create') {
        const newBunker = await bunkerService.createBunker(payload);
        // Show success toast
        navigate(`/bunkers/${newBunker.id}`);
      } else {
        await bunkerService.updateBunker(initialData!.id, payload);
        // Show success toast
        if (onSuccess) onSuccess();
      }
    } catch (error) {
      // Show error toast
      console.error('Failed to save bunker:', error);
    } finally {
      setIsSubmitting(false);
    }
  };

  const handleMapClick = (lat: number, lng: number) => {
    setFormData({
      ...formData,
      latitude: lat.toFixed(6),
      longitude: lng.toFixed(6)
    });
  };

  return (
    <form onSubmit={handleSubmit} className="space-y-6">
      {/* Name Field */}
      <div>
        <label className="block text-sm font-medium text-gray-700 mb-2">
          Bunker Name *
        </label>
        <input
          type="text"
          value={formData.name}
          onChange={(e) => setFormData({ ...formData, name: e.target.value })}
          className="w-full px-4 py-2 border rounded-lg focus:ring-2 focus:ring-blue-500"
          placeholder="e.g., North Bunker A"
        />
        {errors.name && <p className="text-red-500 text-sm mt-1">{errors.name}</p>}
      </div>

      {/* Map Location Picker */}
      <div>
        <label className="block text-sm font-medium text-gray-700 mb-2">
          Location (click map to set)
        </label>
        <MapLocationPicker
          latitude={parseFloat(formData.latitude) || 0}
          longitude={parseFloat(formData.longitude) || 0}
          onLocationSelect={handleMapClick}
        />
        <div className="grid grid-cols-2 gap-4 mt-2">
          <div>
            <input
              type="number"
              step="0.000001"
              value={formData.latitude}
              onChange={(e) => setFormData({ ...formData, latitude: e.target.value })}
              placeholder="Latitude"
              className="w-full px-4 py-2 border rounded-lg"
            />
            {errors.latitude && <p className="text-red-500 text-sm mt-1">{errors.latitude}</p>}
          </div>
          <div>
            <input
              type="number"
              step="0.000001"
              value={formData.longitude}
              onChange={(e) => setFormData({ ...formData, longitude: e.target.value })}
              placeholder="Longitude"
              className="w-full px-4 py-2 border rounded-lg"
            />
            {errors.longitude && <p className="text-red-500 text-sm mt-1">{errors.longitude}</p>}
          </div>
        </div>
      </div>

      {/* Orientation Selector */}
      <div>
        <label className="block text-sm font-medium text-gray-700 mb-2">
          Bunker Orientation (degrees from North)
        </label>
        <OrientationSelector
          value={parseFloat(formData.orientation_degrees)}
          onChange={(deg) => setFormData({ ...formData, orientation_degrees: deg.toString() })}
        />
        {errors.orientation_degrees && (
          <p className="text-red-500 text-sm mt-1">{errors.orientation_degrees}</p>
        )}
      </div>

      {/* Other Fields */}
      <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
        <div>
          <label className="block text-sm font-medium text-gray-700 mb-2">
            Fan Count *
          </label>
          <input
            type="number"
            min="1"
            value={formData.fan_count}
            onChange={(e) => setFormData({ ...formData, fan_count: e.target.value })}
            className="w-full px-4 py-2 border rounded-lg"
          />
          {errors.fan_count && <p className="text-red-500 text-sm mt-1">{errors.fan_count}</p>}
        </div>

        <div>
          <label className="block text-sm font-medium text-gray-700 mb-2">
            Electricity Cost ($/kWh) *
          </label>
          <input
            type="number"
            step="0.01"
            value={formData.electricity_cost_kwh}
            onChange={(e) => setFormData({ ...formData, electricity_cost_kwh: e.target.value })}
            className="w-full px-4 py-2 border rounded-lg"
          />
          {errors.electricity_cost_kwh && (
            <p className="text-red-500 text-sm mt-1">{errors.electricity_cost_kwh}</p>
          )}
        </div>

        <div>
          <label className="block text-sm font-medium text-gray-700 mb-2">
            Fan Power (Watts) *
          </label>
          <input
            type="number"
            value={formData.fan_power_watts}
            onChange={(e) => setFormData({ ...formData, fan_power_watts: e.target.value })}
            className="w-full px-4 py-2 border rounded-lg"
          />
          {errors.fan_power_watts && (
            <p className="text-red-500 text-sm mt-1">{errors.fan_power_watts}</p>
          )}
        </div>
      </div>

      {/* Action Buttons */}
      <div className="flex justify-end space-x-4">
        <button
          type="button"
          onClick={() => navigate(-1)}
          className="px-6 py-2 border border-gray-300 rounded-lg hover:bg-gray-50"
        >
          Cancel
        </button>
        <button
          type="submit"
          disabled={isSubmitting}
          className="px-6 py-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 disabled:opacity-50"
        >
          {isSubmitting ? 'Saving...' : mode === 'create' ? 'Create Bunker' : 'Update Bunker'}
        </button>
      </div>
    </form>
  );
};
```

import { MapContainer, TileLayer, Marker, useMapEvents } from 'react-leaflet';
import 'leaflet/dist/leaflet.css';
import L from 'leaflet';

// Fix Leaflet default marker icon issue in Webpack
delete (L.Icon.Default.prototype as any)._getIconUrl;
L.Icon.Default.mergeOptions({
  iconRetinaUrl: require('leaflet/dist/images/marker-icon-2x.png'),
  iconUrl: require('leaflet/dist/images/marker-icon.png'),
  shadowUrl: require('leaflet/dist/images/marker-shadow.png'),
});

interface MapLocationPickerProps {
  latitude: number;
  longitude: number;
  onLocationSelect: (lat: number, lng: number) => void;
}

function MapClickHandler({ onLocationSelect }: { onLocationSelect: (lat: number, lng: number) => void }) {
  useMapEvents({
    click: (e) => {
      onLocationSelect(e.latlng.lat, e.latlng.lng);
    }
  });
  return null;
}

export const MapLocationPicker: React.FC<MapLocationPickerProps> = ({
  latitude,
  longitude,
  onLocationSelect
}) => {
  const center: [number, number] = [latitude || 41.5, longitude || -93.5]; // Default to Iowa

  return (
    <div className="h-64 w-full rounded-lg overflow-hidden border">
      <MapContainer
        center={center}
        zoom={13}
        style={{ height: '100%', width: '100%' }}
      >
        <TileLayer
          attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a>'
          url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
        />
        <MapClickHandler onLocationSelect={onLocationSelect} />
        {latitude && longitude && (
          <Marker position={[latitude, longitude]} />
        )}
      </MapContainer>
    </div>
  );
};

### Orientation Selector Component

```typescript
// components/bunker/OrientationSelector.tsx
import React, { useRef, useEffect } from 'react';

interface OrientationSelectorProps {
  value: number; // 0-360
  onChange: (degrees: number) => void;
}

export const OrientationSelector: React.FC<OrientationSelectorProps> = ({ value, onChange }) => {
  const canvasRef = useRef<HTMLCanvasElement>(null);

  const drawCompass = () => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    const centerX = canvas.width / 2;
    const centerY = canvas.height / 2;
    const radius = 80;

    // Clear canvas
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    // Draw compass circle
    ctx.strokeStyle = '#cbd5e1';
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.arc(centerX, centerY, radius, 0, 2 * Math.PI);
    ctx.stroke();

    // Draw cardinal directions
    ctx.fillStyle = '#1e293b';
    ctx.font = 'bold 14px sans-serif';
    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillText('N', centerX, centerY - radius - 15);
    ctx.fillText('E', centerX + radius + 15, centerY);
    ctx.fillText('S', centerX, centerY + radius + 15);
    ctx.fillText('W', centerX - radius - 15, centerY);

    // Draw orientation arrow
    const angleRad = ((value - 90) * Math.PI) / 180; // Adjust for canvas coordinates
    const arrowLength = radius - 10;

    ctx.strokeStyle = '#2563eb';
    ctx.fillStyle = '#2563eb';
    ctx.lineWidth = 3;

    // Arrow line
    ctx.beginPath();
    ctx.moveTo(centerX, centerY);
    ctx.lineTo(
      centerX + arrowLength * Math.cos(angleRad),
      centerY + arrowLength * Math.sin(angleRad)
    );
    ctx.stroke();

    // Arrow head
    const headLength = 15;
    const headAngle = Math.PI / 6;
    const endX = centerX + arrowLength * Math.cos(angleRad);
    const endY = centerY + arrowLength * Math.sin(angleRad);

    ctx.beginPath();
    ctx.moveTo(endX, endY);
    ctx.lineTo(
      endX - headLength * Math.cos(angleRad - headAngle),
      endY - headLength * Math.sin(angleRad - headAngle)
    );
    ctx.lineTo(
      endX - headLength * Math.cos(angleRad + headAngle),
      endY - headLength * Math.sin(angleRad + headAngle)
    );
    ctx.closePath();
    ctx.fill();
  };

  useEffect(() => {
    drawCompass();
  }, [value]);

  const handleCanvasClick = (e: React.MouseEvent<HTMLCanvasElement>) => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const rect = canvas.getBoundingClientRect();
    const x = e.clientX - rect.left - canvas.width / 2;
    const y = e.clientY - rect.top - canvas.height / 2;

    let angle = (Math.atan2(y, x) * 180) / Math.PI + 90;
    if (angle < 0) angle += 360;

    onChange(Math.round(angle) % 360);
  };

  return (
    <div className="flex flex-col items-center">
      <canvas
        ref={canvasRef}
        width={200}
        height={200}
        onClick={handleCanvasClick}
        className="cursor-pointer border rounded-lg"
      />
      <div className="mt-2 flex items-center space-x-2">
        <input
          type="number"
          min="0"
          max="359"
          value={value}
          onChange={(e) => onChange(parseInt(e.target.value) || 0)}
          className="w-20 px-3 py-1 border rounded"
        />
        <span className="text-gray-600">degrees</span>
      </div>
    </div>
  );
};
```

### Bunker Service Functions

// services/bunker.service.ts
import api from './api';
import { Bunker, BunkerCreateRequest, BunkerUpdateRequest } from '@/types';

export async function createBunker(data: BunkerCreateRequest): Promise<Bunker> {
  const response = await api.post<Bunker>('/bunkers', data);
  return response.data;
}

export async function updateBunker(id: string, data: BunkerUpdateRequest): Promise<Bunker> {
  const response = await api.put<Bunker>(`/bunkers/${id}`, data);
  return response.data;
}

export async function deleteBunker(id: string): Promise<void> {
  await api.delete(`/bunkers/${id}`);
}

export async function getBunker(id: string): Promise<Bunker> {
  const response = await api.get<Bunker>(`/bunkers/${id}`);
  return response.data;
}

### Toast Notification Pattern

Use a simple toast component for success/error feedback:

```typescript
// components/common/Toast.tsx
import React, { useEffect } from 'react';

interface ToastProps {
  message: string;
  type: 'success' | 'error';
  onClose: () => void;
  duration?: number;
}

export const Toast: React.FC<ToastProps> = ({ message, type, onClose, duration = 3000 }) => {
  useEffect(() => {
    const timer = setTimeout(onClose, duration);
    return () => clearTimeout(timer);
  }, [duration, onClose]);

  const bgColor = type === 'success' ? 'bg-green-500' : 'bg-red-500';

  return (
    <div className={`fixed top-4 right-4 ${bgColor} text-white px-6 py-3 rounded-lg shadow-lg z-50`}>
      <p>{message}</p>
    </div>
  );
};
```
