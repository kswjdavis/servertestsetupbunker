# Story 2.4: Weather API Integration (Backend)

## Status
Draft

## Story
**As a** backend server,
**I want** to fetch wind data from aviation weather.gov API every 60 seconds,
**so that** the control logic engine can make shutdown decisions based on current wind conditions (FR1).

## Acceptance Criteria

1. WeatherService fetches data from weather.gov `/stations/{id}/observations/latest` endpoint
2. User-Agent header included in all requests (required by weather.gov)
3. Wind speed converted from km/h to mph
4. Wind direction in degrees (0-360) stored directly
5. Temperature converted from Celsius to Fahrenheit
6. Latest weather data cached in memory (no historical storage for POC)
7. Background task polls API every 60 seconds
8. Network errors handled gracefully - use last known good data
9. Stale data threshold: log warning if no successful fetch for 10 minutes
10. Unit tests cover API parsing and error handling

### WeatherService Implementation

```python
import httpx
from datetime import datetime, timedelta
from app.schemas.weather import WeatherData
from app.core.config import settings
import logging

logger = logging.getLogger(__name__)

class WeatherService:
    def __init__(self):
        self.base_url = "https://api.weather.gov"
        self.station_id = settings.WEATHER_STATION_ID
        self.cached_weather: WeatherData | None = None
        self.last_fetch: datetime | None = None

    async def fetch_weather_from_api(self) -> WeatherData:
        url = f"{self.base_url}/stations/{self.station_id}/observations/latest"
        headers = {
            "User-Agent": "BunkerColab/1.0 (contact@yourdomain.com)",
            "Accept": "application/geo+json"
        }

        try:
            async with httpx.AsyncClient(timeout=10.0) as client:
                response = await client.get(url, headers=headers)
                response.raise_for_status()
                data = response.json()

                props = data["properties"]

                # Convert units
                wind_speed_kmh = props.get("windSpeed", {}).get("value")
                wind_speed_mph = wind_speed_kmh * 0.621371 if wind_speed_kmh else 0.0

                temp_c = props.get("temperature", {}).get("value")
                temp_f = (temp_c * 9/5 + 32) if temp_c else 32.0

                wind_dir = props.get("windDirection", {}).get("value", 0)

                weather = WeatherData(
                    station_id=self.station_id,
                    wind_speed_mph=wind_speed_mph,
                    wind_direction_degrees=wind_dir,
                    temperature_f=temp_f,
                    fetched_at=datetime.utcnow(),
                    observation_time=datetime.fromisoformat(props["timestamp"].replace("Z", "+00:00"))
                )

                self.cached_weather = weather
                self.last_fetch = datetime.utcnow()
                logger.debug(f"Weather fetched: {wind_speed_mph:.1f} mph @ {wind_dir}°")
                return weather

        except Exception as e:
            logger.error(f"Weather API fetch failed: {e}")
            if self.cached_weather:
                logger.warning("Using cached weather data")
                return self.cached_weather
            raise

    def get_current_weather(self) -> WeatherData:
        if self.is_weather_stale():
            logger.warning("Weather data is stale (>10 minutes old)")

        if not self.cached_weather:
            raise ValueError("No weather data available")

        return self.cached_weather

    def is_weather_stale(self) -> bool:
        if not self.last_fetch:
            return True
        return datetime.utcnow() - self.last_fetch > timedelta(minutes=10)

# Global instance
weather_service = WeatherService()
```

### Background Task (FastAPI Lifespan)

```python
from contextlib import asynccontextmanager
import asyncio

async def weather_polling_task():
    while True:
        try:
            await weather_service.fetch_weather_from_api()
        except Exception as e:
            logger.error(f"Weather polling error: {e}")
        await asyncio.sleep(60)  # 60 seconds

@asynccontextmanager
async def lifespan(app: FastAPI):
    # Startup
    task = asyncio.create_task(weather_polling_task())
    logger.info("Weather polling task started")
    yield
    # Shutdown
    task.cancel()
    logger.info("Weather polling task stopped")

app = FastAPI(lifespan=lifespan)
```

### API Endpoint

```python
@router.get("/weather/current", response_model=WeatherData)
async def get_current_weather(current_user: User = Depends(get_current_user)):
    try:
        return weather_service.get_current_weather()
    except ValueError:
        raise HTTPException(status_code=503, detail="Weather data unavailable")
```
