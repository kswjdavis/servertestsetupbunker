**As a** web UI operator,
**I want** visually appealing fan layout that clearly shows fan positions and status,
**so that** I can quickly understand bunker configuration and identify issues.

### SVG Fan Visualization

[Source: architecture.md#Frontend Architecture - Component Template Example]

```typescript
// components/bunker/FanSVG.tsx
import React from 'react';

interface FanSVGProps {
  position: number;
  status: 'online-on' | 'online-off' | 'offline';
  size?: number;
  onClick?: () => void;
}

export const FanSVG: React.FC<FanSVGProps> = ({
  position,
  status,
  size = 80,
  onClick
}) => {
  const colors = {
    'online-on': '#22c55e',    // Green
    'online-off': '#3b82f6',   // Blue
    'offline': '#ef4444'       // Red
  };

  const color = colors[status];

  return (
    <svg
      width={size}
      height={size}
      viewBox="0 0 100 100"
      onClick={onClick}
      className="cursor-pointer transition-all duration-300 hover:scale-110"
      role="button"
      aria-label={`Fan ${position}, ${status.replace('-', ' ')}`}
      tabIndex={0}
    >
      {/* Fan circle */}
      <circle
        cx="50"
        cy="50"
        r="45"
        fill={color}
        stroke="#1f2937"
        strokeWidth="2"
        className="transition-all duration-500"
      />

      {/* Fan blades (rotating animation for ON status) */}
      <g
        className={status === 'online-on' ? 'animate-spin origin-center' : ''}
        style={{ transformOrigin: '50% 50%' }}
      >
        {[0, 120, 240].map(angle => (
          <path
            key={angle}
            d="M 50 50 L 50 20 Q 60 30 50 40 Z"
            fill="rgba(255, 255, 255, 0.7)"
            transform={`rotate(${angle} 50 50)`}
          />
        ))}
      </g>

      {/* Position number */}
      <text
        x="50"
        y="60"
        textAnchor="middle"
        fontSize="24"
        fontWeight="bold"
        fill="#fff"
      >
        {position}
      </text>

      {/* Status indicator dot */}
      <circle
        cx="80"
        cy="20"
        r="8"
        fill={status === 'offline' ? '#fbbf24' : '#10b981'}
        stroke="#fff"
        strokeWidth="2"
      />
    </svg>
  );
};
```

          </div>
          <div className="flex justify-between">
            <span className="text-gray-400">WiFi Signal:</span>

### Layout Component

```typescript
// components/bunker/FanLayoutVisualization.tsx
import React from 'react';
import { useNavigate } from 'react-router-dom';
import { FanStatus } from '@/types';
import { FanSVG } from './FanSVG';
import { FanTooltip } from './FanTooltip';
import { StatusLegend } from './StatusLegend';

interface FanLayoutVisualizationProps {
  devices: FanStatus[];
  fanCount: number;
}

export const FanLayoutVisualization: React.FC<FanLayoutVisualizationProps> = ({
  devices,
  fanCount
}) => {
  const navigate = useNavigate();

  // Calculate responsive grid columns based on fan count
  const getGridCols = (count: number): string => {
    if (count <= 4) return 'grid-cols-2';
    if (count <= 6) return 'grid-cols-3';
    if (count <= 9) return 'grid-cols-3 lg:grid-cols-3';
    return 'grid-cols-3 lg:grid-cols-4';
  };

  const getFanStatus = (device: FanStatus): 'online-on' | 'online-off' | 'offline' => {
    if (!device.is_online) return 'offline';
    return device.relay_state === 'ON' ? 'online-on' : 'online-off';
  };

  return (
    <div className="bg-white rounded-lg shadow-lg p-6">
      <div className="flex items-center justify-between mb-6">
        <h3 className="text-xl font-semibold">Fan Layout</h3>
        <StatusLegend />
      </div>

      <div className={`grid ${getGridCols(fanCount)} gap-6 justify-items-center`}>
        {Array.from({ length: fanCount }, (_, i) => i + 1).map(position => {
          const device = devices.find(d => d.fan_position === position);

          if (!device) {
            // Empty slot (device not provisioned yet)
            return (
              <div
                key={position}
                className="flex items-center justify-center w-20 h-20 border-2 border-dashed border-gray-300 rounded-full"
                aria-label={`Fan ${position} not provisioned`}
              >
                <span className="text-gray-400 text-sm">#{position}</span>
              </div>
            );
          }

          return (
            <FanTooltip key={device.device_id} fan={device}>
              <FanSVG
                position={position}
                status={getFanStatus(device)}
                onClick={() => navigate(`/devices/${device.device_id}`)}
              />
            </FanTooltip>
          );
        })}
      </div>
    </div>
  );
};
```
