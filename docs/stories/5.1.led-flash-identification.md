```c
#ifndef LED_CONTROLLER_H
#define LED_CONTROLLER_H

#include <stdint.h>
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

/**
 * @brief Initialize LED controller GPIO
 *
 * Configures LED GPIO pin as output.
 * GPIO pin number set via menuconfig (CONFIG_LED_GPIO_PIN).
 *
 * @return ESP_OK on success, ESP_FAIL on error
 */
esp_err_t led_controller_init(void);

/**
 * @brief Start LED flash task
 *
 * Creates a FreeRTOS task that continuously flashes the LED based on
 * the led_flash_sequence value (1-10) stored in NVS.
 *
 * Flash pattern:
 * - N rapid blinks (200ms ON, 200ms OFF)
 * - 2 second pause
 * - Repeat forever
 *
 * @param flash_count Number of blinks (1-10)
 * @return ESP_OK on success, ESP_FAIL on error
 */
esp_err_t led_flash_task_start(uint8_t flash_count);

/**
 * @brief Stop LED flash task (cleanup)
 */
void led_flash_task_stop(void);

/**
 * @brief Set LED state (manual control for testing)
 *
 * @param state true = ON, false = OFF
 */
void led_set_state(bool state);

#endif // LED_CONTROLLER_H
```

**led_controller.c:**
```c
// firmware/main/led_controller.c
#include "led_controller.h"
#include "driver/gpio.h"
#include "esp_log.h"
#include "freertos/FreeRTOS.h"
#include "freertos/task.h"

static const char *TAG = "LED_CTRL";

// LED GPIO pin configured via menuconfig
#ifndef CONFIG_LED_GPIO_PIN
#define CONFIG_LED_GPIO_PIN 2  // Default onboard LED for ESP32-DevKitC
#endif

#define LED_ON_TIME_MS   200   // AC3: 200ms ON
#define LED_OFF_TIME_MS  200   // AC3: 200ms OFF
#define PAUSE_TIME_MS    2000  // AC2: 2 second pause between sequences

static TaskHandle_t led_task_handle = NULL;
static uint8_t flash_sequence_count = 0;

// Initialize LED GPIO (AC5: configurable via menuconfig)
esp_err_t led_controller_init(void) {
    gpio_config_t io_conf = {
        .pin_bit_mask = (1ULL << CONFIG_LED_GPIO_PIN),
        .mode = GPIO_MODE_OUTPUT,
        .pull_up_en = GPIO_PULLUP_DISABLE,
        .pull_down_en = GPIO_PULLDOWN_DISABLE,
        .intr_type = GPIO_INTR_DISABLE,
    };

    esp_err_t ret = gpio_config(&io_conf);
    if (ret == ESP_OK) {
        gpio_set_level(CONFIG_LED_GPIO_PIN, 0);  // Start with LED OFF
        ESP_LOGI(TAG, "LED controller initialized on GPIO %d", CONFIG_LED_GPIO_PIN);
    } else {
        ESP_LOGE(TAG, "Failed to configure LED GPIO");
    }

    return ret;
}

// Set LED state (helper function)
void led_set_state(bool state) {
    gpio_set_level(CONFIG_LED_GPIO_PIN, state ? 1 : 0);
}

// AC1, AC2, AC3: LED flash task
static void led_flash_task(void *pvParameters) {
    uint8_t flash_count = flash_sequence_count;

    ESP_LOGI(TAG, "LED flash task started with sequence count: %d", flash_count);

    while (1) {
        // AC2: Flash N times (flash_count)
        for (int i = 0; i < flash_count; i++) {
            // AC3: 200ms ON
            led_set_state(true);
            vTaskDelay(pdMS_TO_TICKS(LED_ON_TIME_MS));

            // AC3: 200ms OFF
            led_set_state(false);
            vTaskDelay(pdMS_TO_TICKS(LED_OFF_TIME_MS));
        }

        // AC2: 2 second pause before repeating
        vTaskDelay(pdMS_TO_TICKS(PAUSE_TIME_MS));
    }
}

// AC4, AC6: Start LED flash task
esp_err_t led_flash_task_start(uint8_t flash_count) {
    if (flash_count < 1 || flash_count > 10) {
        ESP_LOGE(TAG, "Invalid flash count: %d (must be 1-10)", flash_count);
        return ESP_FAIL;
    }

    flash_sequence_count = flash_count;

    // Stop existing task if running
    if (led_task_handle != NULL) {
        led_flash_task_stop();
    }

    // AC4: Create FreeRTOS task
    BaseType_t ret = xTaskCreate(
        led_flash_task,
        "led_flash_task",
        2048,  // Stack size (AC8: small footprint)
        NULL,
        5,     // Priority (low, doesn't interfere with critical tasks)
        &led_task_handle
    );

    if (ret == pdPASS) {
        ESP_LOGI(TAG, "LED flash task created successfully");
        return ESP_OK;
    } else {
        ESP_LOGE(TAG, "Failed to create LED flash task");
        return ESP_FAIL;
    }
}

// Stop LED flash task (cleanup)
void led_flash_task_stop(void) {
    if (led_task_handle != NULL) {
        vTaskDelete(led_task_handle);
        led_task_handle = NULL;
        led_set_state(false);  // Turn off LED
        ESP_LOGI(TAG, "LED flash task stopped");
    }
}
```

### Integration with Main Application

**main.c (UPDATE):**
```c
// firmware/main/main.c
#include "led_controller.h"
#include "nvs_storage.h"
#include "wifi_manager.h"

static const char *TAG = "MAIN";

void app_main(void) {
    // Initialize NVS
    nvs_flash_init();

    // AC5: Initialize LED GPIO
    led_controller_init();

    // Read device configuration from NVS
    device_config_t config;
    if (nvs_read_device_config(&config) == ESP_OK) {
        ESP_LOGI(TAG, "Device provisioned - LED flash sequence: %d", config.led_flash_sequence);
    } else {
        ESP_LOGW(TAG, "Device not provisioned - using default LED sequence 1");
        config.led_flash_sequence = 1;  // Default flash pattern
    }

    // AC6: Start LED flash after WiFi connection (or immediately for testing)
    // For production: Start after successful WiFi connection
    // For testing: Start immediately to verify flash pattern
    #ifdef CONFIG_LED_FLASH_START_IMMEDIATE
        led_flash_task_start(config.led_flash_sequence);
    #else
        // Wait for WiFi connection, then start LED flash
        wifi_init();
        // In wifi_event_handler, call led_flash_task_start() on CONNECTED event
    #endif

    // Continue with dead-man timer, HTTP client, etc.
    // ...
}
```

**WiFi Event Handler (AC6):**
```c
// firmware/main/wifi_manager.c
static void wifi_event_handler(void* arg, esp_event_base_t event_base,
                                int32_t event_id, void* event_data) {
    if (event_base == WIFI_EVENT && event_id == WIFI_EVENT_STA_START) {
        esp_wifi_connect();
    } else if (event_base == IP_EVENT && event_id == IP_EVENT_STA_GOT_IP) {
        ESP_LOGI(TAG, "WiFi connected, got IP address");

        // AC6: Start LED flash after network connection
        device_config_t config;
        if (nvs_read_device_config(&config) == ESP_OK) {
            led_flash_task_start(config.led_flash_sequence);
        }
    }
}
```

```markdown
## ESP32 LED Flash Identification

Each ESP32 device flashes its onboard LED in a unique pattern to help identify
which physical device corresponds to which database record.

### Flash Pattern
- Device with `led_flash_sequence = 1`: 1 blink, 2s pause, repeat
- Device with `led_flash_sequence = 2`: 2 blinks, 2s pause, repeat
- ...
- Device with `led_flash_sequence = 10`: 10 blinks, 2s pause, repeat

### Blink Timing
- Each blink: 200ms ON, 200ms OFF
- Pause between sequences: 2 seconds

### During Deployment
1. Provision device via Web UI - note the LED flash sequence number
2. Power on ESP32 device
3. Wait for WiFi connection (LED flash starts automatically)
4. Count the LED blinks to identify the device
5. Install device in the correct fan position

### Troubleshooting
- **LED not flashing:** Check WiFi connection, verify device provisioned
- **Wrong flash count:** Re-provision device to reassign LED sequence
- **No LED visible:** Verify GPIO pin matches hardware (default GPIO 2)
```
