### Complete Database Schema (from architecture.md:1594-1738)

**SQL Schema Reference:**
```sql
-- Users table
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    username VARCHAR(50) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    email VARCHAR(255),
    role VARCHAR(20) NOT NULL CHECK (role IN ('admin', 'operator', 'viewer')),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    last_login TIMESTAMP WITH TIME ZONE
);

-- Bunkers table
CREATE TABLE bunkers (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    latitude DOUBLE PRECISION NOT NULL,
    longitude DOUBLE PRECISION NOT NULL,
    orientation_degrees DOUBLE PRECISION NOT NULL CHECK (orientation_degrees >= 0 AND orientation_degrees < 360),
    fan_count INTEGER NOT NULL CHECK (fan_count > 0),
    wind_threshold_mph DOUBLE PRECISION,
    electricity_cost_kwh DOUBLE PRECISION NOT NULL,
    fan_power_watts INTEGER NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Devices table
CREATE TABLE devices (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    auth_token UUID UNIQUE NOT NULL DEFAULT gen_random_uuid(),
    bunker_id UUID NOT NULL REFERENCES bunkers(id) ON DELETE CASCADE,
    fan_position INTEGER NOT NULL,
    mac_address VARCHAR(17) UNIQUE NOT NULL,
    firmware_version VARCHAR(50),
    last_seen TIMESTAMP WITH TIME ZONE,
    provisioned_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    led_flash_sequence INTEGER NOT NULL CHECK (led_flash_sequence BETWEEN 1 AND 10),
    UNIQUE (bunker_id, fan_position)
);

CREATE INDEX idx_devices_bunker_id ON devices(bunker_id);
CREATE INDEX idx_devices_last_seen ON devices(last_seen);

-- Device Status table (current status only, no history for POC)
CREATE TABLE device_status (
    device_id UUID PRIMARY KEY REFERENCES devices(id) ON DELETE CASCADE,
    relay_state VARCHAR(10) NOT NULL CHECK (relay_state IN ('ON', 'OFF')),
    uptime_seconds INTEGER NOT NULL,
    wifi_rssi INTEGER NOT NULL,
    countdown_timer_remaining INTEGER NOT NULL CHECK (countdown_timer_remaining BETWEEN 0 AND 300),
    reported_at TIMESTAMP WITH TIME ZONE NOT NULL,
    server_received_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Time Window Overrides table
CREATE TABLE time_window_overrides (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    bunker_id UUID REFERENCES bunkers(id) ON DELETE CASCADE,  -- NULL means global override
    start_time TIMESTAMP WITH TIME ZONE NOT NULL,
    end_time TIMESTAMP WITH TIME ZONE NOT NULL,
    reason TEXT NOT NULL,
    created_by UUID NOT NULL REFERENCES users(id),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CHECK (end_time > start_time)
);

-- Weather Data table (current conditions only)
CREATE TABLE weather_data (
    station_id VARCHAR(10) PRIMARY KEY,
    wind_speed_mph DOUBLE PRECISION NOT NULL,
    wind_direction_degrees DOUBLE PRECISION NOT NULL CHECK (wind_direction_degrees >= 0 AND wind_direction_degrees < 360),
    temperature_f DOUBLE PRECISION NOT NULL,
    fetched_at TIMESTAMP WITH TIME ZONE NOT NULL,
    observation_time TIMESTAMP WITH TIME ZONE NOT NULL
);

-- Global Config table (singleton)
CREATE TABLE global_config (
    id INTEGER PRIMARY KEY CHECK (id = 1),
    default_wind_threshold_mph DOUBLE PRECISION NOT NULL,
    default_electricity_cost_kwh DOUBLE PRECISION NOT NULL,
    default_fan_power_watts INTEGER NOT NULL,
    weather_station_id VARCHAR(10) NOT NULL,
    weather_poll_interval_seconds INTEGER NOT NULL,
    shutdown_broadcast_interval_seconds INTEGER NOT NULL DEFAULT 60,
    device_offline_threshold_seconds INTEGER NOT NULL DEFAULT 120
);
```

### SQLAlchemy ORM Models Pattern

Use async SQLAlchemy 2.0 style with typed relationships:

```python
from sqlalchemy.ext.asyncio import AsyncAttrs
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, relationship
from sqlalchemy import String, DateTime, ForeignKey
from datetime import datetime
from uuid import UUID, uuid4

class Base(AsyncAttrs, DeclarativeBase):
    pass

class User(Base):
    __tablename__ = "users"

    id: Mapped[UUID] = mapped_column(primary_key=True, default=uuid4)
    username: Mapped[str] = mapped_column(String(50), unique=True, nullable=False)
    password_hash: Mapped[str] = mapped_column(String(255), nullable=False)
    email: Mapped[str | None] = mapped_column(String(255))
    role: Mapped[str] = mapped_column(String(20), nullable=False)
    created_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), default=datetime.utcnow)
    last_login: Mapped[datetime | None] = mapped_column(DateTime(timezone=True))
```

### Repository Pattern Base Class

```python
from typing import Generic, TypeVar, Type
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select

ModelType = TypeVar("ModelType")

class BaseRepository(Generic[ModelType]):
    def __init__(self, model: Type[ModelType], session: AsyncSession):
        self.model = model
        self.session = session

    async def get(self, id):
        result = await self.session.execute(
            select(self.model).where(self.model.id == id)
        )
        return result.scalar_one_or_none()

    async def create(self, **kwargs):
        instance = self.model(**kwargs)
        self.session.add(instance)
        await self.session.commit()
        await self.session.refresh(instance)
        return instance
```
