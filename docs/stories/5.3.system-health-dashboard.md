# Story 5.3: System Health Dashboard

## Status
Ready for Dev

## Story
**As a** system administrator,
**I want** system health dashboard showing overall system status,
**so that** I can quickly identify issues across all bunkers and devices.

## Acceptance Criteria

1. System health page accessible from header
2. Shows: Total devices, Online devices, Offline devices, System uptime
3. Recent alerts/warnings list (auth failures, stale weather, offline devices)
4. Weather service status (last successful fetch, current conditions)
5. Database connection status
6. Backend service uptime
7. Aggregate energy savings (all bunkers)
8. Color-coded health indicators (green, yellow, red)
9. Poll every 10 seconds
10. Export system status as JSON or CSV

### Backend Implementation (AC2-AC8)

**system.py (NEW):**
```python
# app/api/system.py
from fastapi import APIRouter, Depends
from app.services.system_health_service import SystemHealthService

router = APIRouter(prefix="/system", tags=["system"])

@router.get("/health")
async def get_system_health(db: AsyncSession = Depends(get_db)):
    service = SystemHealthService(db)
    return await service.get_health_metrics()
```

**SystemHealthService:**
```python
# app/services/system_health_service.py
class SystemHealthService:
    async def get_health_metrics(self):
        device_repo = DeviceRepository(self.db)
        bunker_repo = BunkerRepository(self.db)

        all_devices = await device_repo.list()
        online_devices = [d for d in all_devices if d.is_online]

        weather_status = await self.check_weather_service()

        return {
            "total_devices": len(all_devices),                # AC2
            "online_devices": len(online_devices),            # AC2
            "offline_devices": len(all_devices) - len(online_devices),  # AC2
            "system_uptime_seconds": self.get_uptime(),      # AC2
            "weather_service_status": weather_status,         # AC4
            "database_status": "connected",                   # AC5
            "backend_uptime_seconds": self.get_uptime(),     # AC6
            "total_energy_saved_kwh": await self.calc_savings(),  # AC7
            "alerts": await self.get_alerts(),                # AC3
            "overall_status": self.compute_status(...)       # AC8
        }
```

### Frontend Implementation (AC1, AC9, AC10)

**SystemHealthPage.tsx:**
```typescript
// web/src/pages/SystemHealthPage.tsx
import React from 'react';
import { usePoll } from '@/hooks/usePoll';
import * as systemService from '@/services/system.service';

export const SystemHealthPage: React.FC = () => {
  const [health, setHealth] = useState<SystemHealth | null>(null);

  const fetchHealth = useCallback(async () => {
    const data = await systemService.getSystemHealth();
    setHealth(data);
  }, []);

  // AC9: Poll every 10 seconds
  usePoll(fetchHealth, 10000);

  const exportJSON = () => {
    const json = JSON.stringify(health, null, 2);
    downloadFile(json, 'system-health.json', 'application/json');
  };

  const exportCSV = () => {
    const csv = convertToCSV(health);
    downloadFile(csv, 'system-health.csv', 'text/csv');
  };

  return (
    <div className="container mx-auto px-4 py-8">
      <h1 className="text-3xl font-bold mb-6">System Health</h1>

      {/* AC2: Device metrics */}
      <div className="grid grid-cols-3 gap-4 mb-8">
        <MetricCard title="Total Devices" value={health.total_devices} />
        <MetricCard title="Online" value={health.online_devices} color="green" />
        <MetricCard title="Offline" value={health.offline_devices} color="red" />
      </div>

      {/* AC4-AC6: Service status */}
      <ServiceStatusSection health={health} />

      {/* AC3: Alerts */}
      <AlertsList alerts={health.alerts} />

      {/* AC10: Export buttons */}
      <div className="mt-8 flex gap-4">
        <button onClick={exportJSON} className="btn-secondary">Export JSON</button>
        <button onClick={exportCSV} className="btn-secondary">Export CSV</button>
      </div>
    </div>
  );
};
```
