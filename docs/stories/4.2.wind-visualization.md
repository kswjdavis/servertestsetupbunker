### Wind Direction Calculation

Calculate wind direction relative to bunker orientation:

```typescript
// utils/windCalculations.ts

/**
 * Calculate relative wind direction
 * @param windDegrees - Absolute wind direction (0-360, North = 0)
 * @param bunkerOrientation - Bunker orientation (0-360, North = 0)
 * @returns Relative wind direction in degrees
 */
export function calculateRelativeWind(
  windDegrees: number,
  bunkerOrientation: number
): number {
  let relative = windDegrees - bunkerOrientation;

  // Normalize to 0-360 range
  if (relative < 0) relative += 360;
  if (relative >= 360) relative -= 360;

  return relative;
}

/**
 * Convert degrees to cardinal direction
 * @param degrees - Direction in degrees (0-360)
 * @returns Cardinal direction (N, NE, E, SE, S, SW, W, NW)
 */
export function degreesToCardinal(degrees: number): string {
  const directions = ['N', 'NE', 'E', 'SE', 'S', 'SW', 'W', 'NW'];
  const index = Math.round(degrees / 45) % 8;
  return directions[index];
}
```

### Component Implementation

[Source: architecture.md#Frontend Architecture - Component Template Example]

```typescript
// components/bunker/WindIndicator.tsx
import React from 'react';
import { useWeather } from '@/hooks/useWeather';
import { Bunker, WeatherData } from '@/types';
import { calculateRelativeWind, degreesToCardinal } from '@/utils/windCalculations';
import { WindArrow } from './WindArrow';
import { CompassRose } from './CompassRose';

interface WindIndicatorProps {
  bunker: Bunker;
  className?: string;
}

export const WindIndicator: React.FC<WindIndicatorProps> = ({ bunker, className }) => {
  const { data: weather, loading, error } = useWeather(3000); // Poll every 3 seconds

  if (loading) return <div className="text-gray-500">Loading weather...</div>;
  if (error || !weather) return <div className="text-red-500">Weather unavailable</div>;

  const threshold = bunker.wind_threshold_mph || 15; // Use bunker or global default
  const meetsThreshold = weather.wind_speed_mph >= threshold;
  const relativeWindDegrees = calculateRelativeWind(
    weather.wind_direction_degrees,
    bunker.orientation_degrees
  );

  return (
    <div className={`bg-white rounded-lg shadow-lg p-6 ${className}`}>
      <h3 className="text-lg font-semibold mb-4">Wind Conditions</h3>

      {/* Wind Speed Display */}
      <div className="flex items-center justify-between mb-4">
        <span className="text-gray-700">Wind Speed:</span>
        <span className={`text-2xl font-bold ${meetsThreshold ? 'text-green-600' : 'text-red-600'}`}>
          {weather.wind_speed_mph.toFixed(1)} mph
        </span>
      </div>

      {/* Threshold Indicator */}
      <div className="flex items-center justify-between mb-6">
        <span className="text-gray-700">Threshold:</span>
        <span className="text-lg">{threshold} mph</span>
      </div>

      {/* Status Badge */}
      <div className={`px-4 py-2 rounded-md text-center mb-6 ${
        meetsThreshold ? 'bg-green-100 text-green-800' : 'bg-red-100 text-red-800'
      }`}>
        {meetsThreshold ? 'Shutdown Conditions Met' : 'Fans Must Run'}
      </div>

      {/* Visual Display: Compass + Wind Arrow */}
      <div className="relative flex items-center justify-center" style={{ minHeight: '200px' }}>
        <CompassRose orientation={bunker.orientation_degrees} />
        <WindArrow
          degrees={relativeWindDegrees}
          speed={weather.wind_speed_mph}
          className="absolute"
        />
      </div>

      {/* Wind Direction Info */}
      <div className="text-center mt-4 text-sm text-gray-600">
        Wind from {degreesToCardinal(weather.wind_direction_degrees)}
        ({weather.wind_direction_degrees}°)
        <br />
        Relative to bunker: {relativeWindDegrees.toFixed(0)}°
      </div>
    </div>
  );
};
```

### SVG Wind Arrow Component

```typescript
// components/bunker/WindArrow.tsx
import React from 'react';

interface WindArrowProps {
  degrees: number;      // Relative wind direction
  speed: number;        // Wind speed (for animation)
  className?: string;
}

export const WindArrow: React.FC<WindArrowProps> = ({ degrees, speed, className }) => {
  // Arrow length based on wind speed
  const arrowLength = Math.min(50 + speed * 2, 120);

  return (
    <svg
      width="200"
      height="200"
      viewBox="0 0 200 200"
      className={className}
      style={{ transform: `rotate(${degrees}deg)` }}
    >
      {/* Arrow shaft */}
      <line
        x1="100"
        y1="100"
        x2="100"
        y2={100 - arrowLength}
        stroke="#2563eb"
        strokeWidth="4"
        strokeLinecap="round"
      />

      {/* Arrow head */}
      <polygon
        points={`100,${100 - arrowLength - 15} 90,${100 - arrowLength} 110,${100 - arrowLength}`}
        fill="#2563eb"
      />

      {/* Center dot */}
      <circle cx="100" cy="100" r="5" fill="#1e40af" />
    </svg>
  );
};
```

### Custom Weather Hook

[Source: architecture.md#Frontend Architecture - Component Organization]

```typescript
// hooks/useWeather.ts
import { useState, useEffect } from 'react';
import { WeatherData } from '@/types';
import * as weatherService from '@/services/weather.service';

export function useWeather(pollInterval: number = 3000) {
  const [data, setData] = useState<WeatherData | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  useEffect(() => {
    const fetchWeather = async () => {
      try {
        const weather = await weatherService.getCurrentWeather();
        setData(weather);
        setError(null);
      } catch (err) {
        setError(err as Error);
      } finally {
        setLoading(false);
      }
    };

    fetchWeather(); // Initial fetch
    const interval = setInterval(fetchWeather, pollInterval);

    return () => clearInterval(interval);
  }, [pollInterval]);

  return { data, loading, error };
}
```
